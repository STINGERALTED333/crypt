local function encodeDrive(id: string)
    local len = #id
    assert(len <= 64, "vehicle id too long")
    local b = buffer.create(3 + len)
    buffer.writeu8(b, 0, 43)
    buffer.writeu16(b, 1, len)
    buffer.writestring(b, 3, id, len)
    return b
end

end

local held, sentAt = {}, {}
local function sendKey(code, down)
    sentAt[code] = os.clock()
    if USE_KEYPRESS and keypress and keyrelease then
        pcall(down and keypress or keyrelease, VK[code])
    else
        pcall(function() VirtualInput:SendKeyEvent(down, code, false, game) end)
    end
end
local function setKey(code, down)
    if (held[code] or false) == down then return end
    held[code] = down
    sendKey(code, down)
end

local DRIVE_KEYS = { KC.W, KC.A, KC.S, KC.D }
local function refreshKeys()
    local now = os.clock()
    for _, k in ipairs(DRIVE_KEYS) do
        local age = now - (sentAt[k] or 0)
        if held[k] then
            if age > KEY_REFRESH then sendKey(k, true) end
        elseif age > 1 then
            sendKey(k, false)
        end
    end
end
local function releaseAll()

end

    if not part then return nil end
    local m = part
    while m do
        if m:IsA("Model") and m:GetAttribute("OwnerUserId") and m:FindFirstChild("Seats") then return m end
        m = m.Parent
    end

end
local function countSeats(part)
    local model = findCarModel(part)
    if not model then return 0, 0 end
    local seats = model:FindFirstChild("Seats")
    if not seats then return 0, 0 end
    local total, occupied = 0, 0
    for _, seat in ipairs(seats:GetChildren()) do
        if seat:IsA("Seat") or seat:IsA("VehicleSeat") then
            total = total + 1
            if seat.Occupant then occupied = occupied + 1 end
        end
    end
    return total, occupied
end

local function flat(v) return Vector3.new(v.X, 0, v.Z) end
local function pt(list, i) local p = list[i]; return Vector3.new(p.x, p.y, p.z) end
local function unitOr(v, fallback) return v.Magnitude > 0.01 and v.Unit or fallback end

local function clean(list)
    local out = {}
    if type(list) ~= "table" then return out end
    for i, p in ipairs(list) do
        if i > MAX_POINTS then break end
        if type(p) == "table" and tonumber(p.x) and tonumber(p.y) and tonumber(p.z) then
            out[#out + 1] = { x = tonumber(p.x), y = tonumber(p.y), z = tonumber(p.z), s = math.clamp(tonumber(p.s) or 0, 0, 500) }
        end
    end
    return out
end
local function distToSegment(p, a, b)
    local abx, abz = b.x - a.x, b.z - a.z
    local len2 = abx * abx + abz * abz
    if len2 == 0 then return math.sqrt((p.x - a.x) ^ 2 + (p.z - a.z) ^ 2) end
    local t = math.clamp(((p.x - a.x) * abx + (p.z - a.z) * abz) / len2, 0, 1)
    return math.sqrt((p.x - (a.x + t * abx)) ^ 2 + (p.z - (a.z + t * abz)) ^ 2)
end
local function simplify(list, eps)
    local n = #list
    if n <= 2 then return list end
    local keep = { [1] = true, [n] = true }
    local stack = { { 1, n } }
    while #stack > 0 do
        local seg = table.remove(stack)
        local i1, i2 = seg[1], seg[2]
        local maxD, maxI = 0, nil
        for i = i1 + 1, i2 - 1 do
            local d = distToSegment(list[i], list[i1], list[i2])
            if d > maxD then maxD, maxI = d, i end
        end
        if maxI and maxD > eps then
            keep[maxI] = true
            stack[#stack + 1] = { i1, maxI }
            stack[#stack + 1] = { maxI, i2 }
        end
    end
    local out = {}
    for i = 1, n do if keep[i] then out[#out + 1] = list[i] end end
    return out
end
local function smoothDrive(list)
    local n = #list
    if n < 5 then return list end
    local len = 0
    for i = 2, n do len += (Vector3.new(list[i].x - list[i - 1].x, 0, list[i].z - list[i - 1].z)).Magnitude end
    local spacing = len / math.max(n - 1, 1)
    local radius = math.floor(5 / math.max(spacing, 0.1) + 0.5)
    if radius < 1 then return list end
    local out = {}
    for i = 1, n do
        local p = list[i]
        if i == 1 or i == n or (p.s or 0) < WAIT_TRIGGER_SPEED then
            out[i] = p
        else
            local sx, sy, sz, ss, c = 0, 0, 0, 0, 0
            local sawStop = false
            for j = math.max(1, i - radius), math.min(n, i + radius) do
                local q = list[j]
                if (q.s or 0) < WAIT_TRIGGER_SPEED then sawStop = true; break end
                sx += q.x; sy += q.y; sz += q.z; ss += q.s; c += 1
            end
            if sawStop or c == 0 then out[i] = p
            else out[i] = { x = sx / c, y = sy / c, z = sz / c, s = ss / c } end
        end
    end
    return out
end
local function parsePath(text)
    text = text:gsub("^\239\187\191", "")
    local ok, data = pcall(HttpService.JSONDecode, HttpService, text)
    if not ok or type(data) ~= "table" then return nil end
    local drive = smoothDrive(clean(data.drive or data.points))
    if #drive < 2 then return nil end
    local walk = simplify(clean(data.walk), WALK_SIMPLIFY)
    local cum = { 0 }
    for i = 2, #drive do cum[i] = cum[i - 1] + (flat(pt(drive, i)) - flat(pt(drive, i - 1))).Magnitude end
    return { drive = drive, walk = walk, cum = cum, exits = ALWAYS_EXIT_AT_END or data.exited == true or #walk > 0 }
end

local httpRequest = (syn and syn.request) or (http and http.request) or http_request or request or (fluxus and fluxus.request)
local lastHttpStatus = "no http function"
local function httpGet(url)
    if httpRequest then
        local ok, res = pcall(httpRequest, { Url = url, Method = "GET" })
        if ok and res then
            lastHttpStatus = "HTTP " .. tostring(res.StatusCode)
            if res.StatusCode == 200 then return res.Body end
            return nil
        end
    end
    local ok, body = pcall(game.HttpGet, game, url)
    if ok and type(body) == "string" then lastHttpStatus = "HTTP 200"; return body end
    return nil
end
local repoPaths, nextProbe = {}, 1
local function refreshRepo()
    local misses, n = 0, nextProbe
    while misses < MISS_LIMIT and n <= MAX_FILES do
        local body = httpGet(BASE_URL .. "/" .. n .. FILE_EXT)
        local p = body and parsePath(body)
        if p then repoPaths[n] = p; nextProbe = n + 1; misses = 0 else misses += 1 end
        n += 1
    end
end
local function loadRandomPath()
    refreshRepo()
    local keys = {}
    for k in pairs(repoPaths) do keys[#keys + 1] = k end
    if #keys > 0 then local k = keys[math.random(#keys)] return repoPaths[k], k .. FILE_EXT end
    if ALLOW_LOCAL_FALLBACK then
        local ok, text = pcall(readfile, LOCAL_FALLBACK)
        local p = ok and parsePath(text)
        if p then return p, LOCAL_FALLBACK end
    end
    return nil, nil, ("Repo failed (%s)"):format(lastHttpStatus)
end

local function exitCar()
    phase = "exiting"
    task.spawn(function()
        local part0 = getVehiclePart()
        local v0 = part0 and flat(part0.AssemblyLinearVelocity) or Vector3.zero
        local dir0 = v0.Magnitude > 1 and v0.Unit or nil
        local function brake()
            local t0 = os.clock()
            while phase == "exiting" and os.clock() - t0 < 4 do
                local v = getVehiclePart() and flat(getVehiclePart().AssemblyLinearVelocity) or Vector3.zero
                if (dir0 and v:Dot(dir0) or 0) < 2 then break end
                setKey(KC.W, false); setKey(KC.A, false); setKey(KC.D, false); setKey(KC.S, true)
                task.wait(0.03)
            end
            releaseAll()
            local t1 = os.clock()
            while phase == "exiting" and os.clock() - t1 < 1.5 do
                local v = getVehiclePart() and flat(getVehiclePart().AssemblyLinearVelocity).Magnitude or 0
                if v < 1.5 then break end
                task.wait(0.05)
            end
        end
        for _ = 1, 3 do
            if phase ~= "exiting" then return end
            brake()
            tapKey(EXIT_KEY)
            task.wait(1.5)
            if not isInCar() then break end
        end
        if phase ~= "exiting" then return end
        if #path.walk == 0 then return stop("Finished (no walk path)") end
        widx, stuckT, walkTarget = 1, 0, nil
        phase = "walk"
    end)
end

local userStopped = false
local function start()
    if phase ~= "idle" then return false end
    phase = "loading"
    label.Text = "Loading paths..."
    local p, name, err = loadRandomPath()
    if phase ~= "loading" then return false end
    if not p then stop(err or "No path found") return false end
    local part = getVehiclePart()
    if not part then stop("No vehicle found") return false end
    path, pathName = p, name
    local pos = part.Position
    local best, bestD = 1, math.huge
    for i = 1, #path.drive do
        local d = (flat(pt(path.drive, i)) - flat(pos)).Magnitude
        if d < bestD then best, bestD = i, d end
    end
    idx = best
    stallT, waitT, steerSmooth, yawRate = 0, 0, 0, 0
    driveState = "normal"; recovery.phase = "none"; recovery.timer = 0; recovery.cycle = 0
    phase = "drive"
    return true
end

local function drivePWM(v, posKey, negKey)
    local phasePos = (os.clock() % PWM_PERIOD) / PWM_PERIOD
    local on = math.abs(v) > phasePos
    setKey(posKey, v > 0 and on)
    setKey(negKey, v < 0 and on)
end

local function driveStep(dt)
    local pts = path.drive
    local part = getVehiclePart()
    if not part then return stop("Lost vehicle") end

    local pos = part.Position
    local vel = flat(part.AssemblyLinearVelocity)
    local speed = vel.Magnitude
    local idxC = math.min(idx, #pts)
    local curV = pts[idxC].s or 0
    local distCur = (flat(pt(pts, idxC)) - flat(pos)).Magnitude

    local distLast = (flat(pt(pts, #pts)) - flat(pos)).Magnitude
    if distLast < EXIT_RADIUS and (path.cum[#pts] - path.cum[idxC]) < EXIT_RADIUS * 3 then
        releaseAll()
        if path.exits then return exitCar() end
        return stop("Finished")
    end

    if curV < WAIT_TRIGGER_SPEED and distCur < WAIT_TRIGGER_RADIUS then
        if speed > 1 then
            drivePWM(-0.7, KC.W, KC.S)
            return
        end
        if WAIT_UNTIL_CAR_FULL then
            local total, occupied = countSeats(part)
            if total == 0 or occupied >= total then
                while idx <= #pts and (pts[idx].s or 0) < WAIT_TRIGGER_SPEED do idx = idx + 1 end
                idxC = math.min(idx, #pts)
                curV = pts[idxC].s or 0
                distCur = (flat(pt(pts, idxC)) - flat(pos)).Magnitude
                waitT = 0
            else
                waitT = waitT + dt
                releaseAll()
                label.Text = ("WAIT PAWNS %d/%d @ wp %d | %.1fs"):format(occupied, total, idx, waitT)
                if waitT > MAX_PAWN_WAIT then
                    while idx <= #pts and (pts[idx].s or 0) < WAIT_TRIGGER_SPEED do idx = idx + 1 end
                    idxC = math.min(idx, #pts)
                    curV = pts[idxC].s or 0
                    distCur = (flat(pt(pts, idxC)) - flat(pos)).Magnitude
                    waitT = 0
                else return end
            end
        else
            waitT = waitT + dt
            if waitT >= WAIT_PER_WAYPOINT then waitT = 0; idx = idx + 1 end
            releaseAll()
            label.Text = ("WAIT %.1fs @ wp %d"):format(waitT, idx)
            return
        end
    end
    waitT = 0

    -- Advance idx (with fix to skip ahead if we are too far from a slow waypoint)
    local lookahead = math.min(LOOKAHEAD_MIN + speed * LOOKAHEAD_SPEED, LOOKAHEAD_MAX)
    while idx < #pts do
        local s = pts[idx].s or 0
        local d = (flat(pt(pts, idx)) - flat(pos)).Magnitude
        if s < WAIT_TRIGGER_SPEED then
            if idx < #pts then
                local nextD = (flat(pt(pts, idx + 1)) - flat(pos)).Magnitude
                if nextD < d then idx = idx + 1; continue end
            end
            break
        end
        if d < lookahead or (speed < 5 and d < 25) then 
            idx = idx + 1 
        else 
            break 
        end
    end
    idxC = math.min(idx, #pts)
    curV = pts[idxC].s or 0
    distCur = (flat(pt(pts, idxC)) - flat(pos)).Magnitude

    local dirVec = flat(pt(pts, idxC) - pos)
    if dirVec.Magnitude < 0.1 then return end
    local dir = dirVec.Unit
    local along = vel:Dot(dir)

    -- Calculate targetSpeed early so stall logic can use it
    local targetSpeed = curV * SPEED_SCALE
    -- If we are far from a slow waypoint, look ahead for the next real speed to accelerate properly
    if targetSpeed < 10 and distCur > 5 then
        local nextS = 0
        for j = idxC + 1, math.min(idxC + 15, #pts) do
            local s = pts[j].s or 0
            if s > targetSpeed then nextS = s; break end
        end
        if nextS > 0 then
            local t = math.clamp((distCur - 5) / 25, 0.3, 1)
            targetSpeed = math.max(targetSpeed, nextS * t * SPEED_SCALE)
        end
    end

    -- Stall detection
    if driveState == "normal" then
        if speed < STALL_SPEED and (targetSpeed > 5 or distCur > 10) then
            stallT = stallT + dt
            if stallT > STALL_TIME then
                driveState = "recovering"
                recovery.phase = "reverse"; recovery.timer = REVERSE_TIME
                recovery.cycle = 1; recovery.steerDir = 1; stallT = 0
            end
        else stallT = 0 end
    end

    if driveState == "recovering" then
        recovery.timer = recovery.timer - dt
        if recovery.phase == "reverse" then
            setKey(KC.W, false); setKey(KC.S, true)
            setKey(KC.A, recovery.steerDir > 0); setKey(KC.D, recovery.steerDir < 0)
            label.Text = ("RECOVER %d REV %.1fs"):format(recovery.cycle, math.max(0, recovery.timer))
        else
            -- FORWARD RECOVERY: steer towards waypoint to pathfind out of the stuck spot
            setKey(KC.S, false)
            local fwd = flat(part.CFrame.LookVector)
            local angle = math.atan2(fwd:Cross(dir).Y, fwd:Dot(dir))
            setKey(KC.A, angle > 0.1)
            setKey(KC.D, angle < -0.1)
            setKey(KC.W, true) 
            label.Text = ("RECOVER %d FWD %.1fs"):format(recovery.cycle, math.max(0, recovery.timer))
        end
        dbg.Text = ("stuck wp %d | speed %.1f"):format(idxC, speed)
        
        if recovery.timer <= 0 then
            if recovery.phase == "reverse" then
                recovery.phase = "forward"; recovery.timer = FORWARD_TIME
            else
                if speed > STALL_SPEED * 2 then driveState = "normal"; recovery.cycle = 0
                elseif recovery.cycle >= MAX_RECOVERY_CYCLES then
                    idx = math.min(idx + SKIP_AHEAD, #pts)
                    driveState = "normal"; recovery.cycle = 0; stallT = 0
                else
                    recovery.cycle = recovery.cycle + 1
                    recovery.phase = "reverse"; recovery.timer = REVERSE_TIME
                    recovery.steerDir = -recovery.steerDir
                end
            end
        end
        return
    end

    -- Forward direction (normal)
    local forward, src
    if fwdLocal then
        forward = flat(part.CFrame:VectorToWorldSpace(fwdLocal))
        forward = forward.Magnitude > 0.01 and forward.Unit or nil
        src = "cf"
        if forward and along > 6 and forward:Dot(vel.Unit) < 0.5 then forward, src = nil, nil end
    end
    if not forward then
        if along > 3 then forward, src = vel.Unit, "vel"
        else forward, src = unitOr(flat(part.CFrame.LookVector), dir), "look" end
    end

    -- Yaw + steering
    local rawYaw = part.AssemblyAngularVelocity.Y
    if along > 8 and math.abs(rawYaw) < 0.3 then
        local rel = part.CFrame:VectorToObjectSpace(vel.Unit)
        local flatRel = Vector3.new(rel.X, 0, rel.Z)
        if flatRel.Magnitude > 0.7 then
            local u = flatRel.Unit
            fwdLocal = fwdLocal and unitOr(fwdLocal:Lerp(u, 0.1), u) or u
        end
    end
    yawRate = yawRate * 0.7 + rawYaw * 0.3
    local angle = math.atan2(forward:Cross(dir).Y, forward:Dot(dir))
    local steerRaw = angle / STEER_FULL_ANGLE - STEER_LEAD * yawRate
    steerSmooth = steerSmooth * STEER_SMOOTH + steerRaw * (1 - STEER_SMOOTH)
    steerSmooth = math.clamp(steerSmooth, -1, 1)
    if math.abs(steerSmooth) < STEER_DEADBAND then steerSmooth = 0 end

    -- Throttle
    local speedErr = targetSpeed - speed
    local throttle
    if speedErr > 0 then throttle = math.clamp(speedErr / SPEED_RANGE, 0, 1)
    elseif along > 1 then throttle = math.clamp(speedErr / SPEED_RANGE, -1, 0)
    elseif along < -1 then throttle = math.clamp(-along / SPEED_RANGE, 0, 1)
    else throttle = 0 end

    -- === TABBED OUT FIX ===
    if not UserInput.WindowFocused then
        setKey(KC.A, steerSmooth > 0)
        setKey(KC.D, steerSmooth < 0)
        setKey(KC.W, throttle > 0.1)
        setKey(KC.S, throttle < -0.1)
    else
        drivePWM(steerSmooth, KC.A, KC.D)
        drivePWM(throttle, KC.W, KC.S)
    end

    local keys = (held[KC.W] and "W" or "-") .. (held[KC.A] and "A" or "-") .. (held[KC.S] and "S" or "-") .. (held[KC.D] and "D" or "-")
    label.Text = ("DRIVE %d/%d | spd %.0f/%.0f | al %+.0f | str %+.2f | %s"):format(idx, #pts, speed, targetSpeed, along, steerSmooth, keys)
    dbg.Text = ("heading:%s  thr:%+.2f  curV:%.1f  dist:%.1f"):format(src, throttle, curV, distCur)
end

local function walkStep(dt)
    local hum, root = getHumanoid(), getRoot()
    if not hum or not root then return end
    local walkPts = path.walk
    local pos = root.Position
    while widx <= #walkPts do
        local reach = (widx == #walkPts) and 2.5 or WALK_REACH
        if (flat(pt(walkPts, widx)) - flat(pos)).Magnitude < reach then widx += 1; walkTarget = nil
        else break end
    end
    if widx > #walkPts then return stop("Arrived at destination") end
    local target = pt(walkPts, widx)
    if target ~= walkTarget or os.clock() - lastMoveTo > 1 then
        hum:MoveTo(target)
        if target ~= walkTarget then stuckT = 0 end
        walkTarget, lastMoveTo = target, os.clock()
    end
    if flat(root.AssemblyLinearVelocity).Magnitude < 1 then
        stuckT += dt
        if stuckT > 1.5 then hum.Jump = true; stuckT = 0 end
    else stuckT = 0 end
    label.Text = ("WALK %d/%d | %s"):format(widx, #walkPts, tostring(pathName))
end

local function wantSprint()
    if not ALWAYS_SPRINT_OUT_OF_CAR then return false end
    if phase == "drive" or phase == "exiting" then return false end
    return getHumanoid() ~= nil and not isInCar()
end

RunService.Heartbeat:Connect(function(dt)
    setKey(KC.LeftShift, wantSprint())
    if phase == "drive" then driveStep(dt); refreshKeys()
    elseif phase == "exiting" then refreshKeys()
    elseif phase == "walk" then walkStep(dt); refreshKeys() end
end)

            task.wait(2)
        end

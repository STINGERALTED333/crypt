local function encodeDrive(id: string)
    local len = #id
    assert(len <= 64, "vehicle id too long")
    local b = buffer.create(3 + len)
    buffer.writeu8(b, 0, 43)
    buffer.writeu16(b, 1, len)
    buffer.writestring(b, 3, id, len)
    return b
end

local function SpawnAsCar()
    repeat task.wait(1) until game.Players.LocalPlayer.Character

    local Car = workspace.VehicleViews:FindFirstChildOfClass("Model")
    local ID = Car and Car:GetAttribute("GarageVehicleId")
    if ID then
        APP_RELIABLE:FireServer(encodeDrive(ID), {})  -- {} = instance list, empty here
        return true
    end

    warn("[ERROR]: No car found in garage!")
    return nil
end

local LOOKAHEAD_MIN    = 8       -- studs ahead to aim at (short = follows corners closely)

local STEER_LEAD       = 2.0     -- seconds-ish of lead: releases steering early based on how fast the car is already turning

local SPEED_RANGE      = 8
local PWM_PERIOD       = 0.12
local KEY_REFRESH      = 0.1    -- seconds between re-sending keydown for held drive keys
local STALL_SPEED      = 2      -- studs/s; below this counts as stopped
local STALL_TIME       = 1.5    -- seconds stopped before throttle is forced to full
-- Walking
local WALK_SIMPLIFY    = 3      -- studs; recorded walk is straightened into lines (bigger = straighter)
local WALK_REACH       = 3      -- studs; waypoint counts as reached
local USE_KEYPRESS     = false  -- true = executor keypress/keyrelease instead of VirtualInputManager
local MAX_POINTS       = 40000

-- Key output (only sends on state change)
local held, sentAt = {}, {}
local function sendKey(code, down)
    sentAt[code] = os.clock()
    if USE_KEYPRESS and keypress and keyrelease then
        pcall(down and keypress or keyrelease, VK[code])
    else
        pcall(function() VirtualInput:SendKeyEvent(down, code, false, game) end)
    end
end
local function setKey(code, down)
    if (held[code] or false) == down then return end
    held[code] = down
    sendKey(code, down)
end

-- The game can drop key state (window focus change, you pressing keys yourself), which would
-- leave us "holding" a key that is really up. So drive keys are re-sent periodically:
-- keydown every KEY_REFRESH seconds for keys we hold, keyup every second for keys we don't.
local DRIVE_KEYS = { KC.W, KC.A, KC.S, KC.D }
local function refreshKeys()
    local now = os.clock()
    for _, k in ipairs(DRIVE_KEYS) do
        local age = now - (sentAt[k] or 0)
        if held[k] then
            if age > KEY_REFRESH then sendKey(k, true) end
        elseif age > 1 then
            sendKey(k, false)
        end
    end
end
local function releaseAll()
    for _, k in ipairs({ KC.W, KC.A, KC.S, KC.D }) do setKey(k, false) end
end
local function tapKey(code)
    setKey(code, true)
    task.wait(0.1)
    setKey(code, false)
end

local function getVehiclePart()
    local hum = getHumanoid()
    local seat = hum and hum.SeatPart
    if seat then return seat.AssemblyRootPart or seat end
    local views = workspace:FindFirstChild("VehicleViews")
    local m = views and views:FindFirstChildOfClass("Model")
    return m and (m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart", true))
end

-- Are we in a car? Works for seat-based and seatless cars:
-- seated, or the character is welded to / parented inside something that is not the character.
local function isInCar()
    local hum, root = getHumanoid(), getRoot()
    if not hum or not root then return false end
    if hum.SeatPart or hum.Sit then return true end
    local rp = root.AssemblyRootPart
    if rp and rp ~= root and not rp:IsDescendantOf(LocalPlayer.Character) then return true end
    local views = workspace:FindFirstChild("VehicleViews")
    if views and root:IsDescendantOf(views) then return true end
    return false
end

local function flat(v) return Vector3.new(v.X, 0, v.Z) end
local function pt(list, i)
    local p = list[i]
    return Vector3.new(p.x, p.y, p.z)
end

----------------------------------------------------------------
-- Path parsing (files are treated as data only: JSON-decoded and validated, never executed)
----------------------------------------------------------------
local function clean(list)
    local out = {}
    if type(list) ~= "table" then return out end
    for i, p in ipairs(list) do
        if i > MAX_POINTS then break end
        if type(p) == "table" and tonumber(p.x) and tonumber(p.y) and tonumber(p.z) then
            out[#out + 1] = {
                x = tonumber(p.x), y = tonumber(p.y), z = tonumber(p.z),
                s = math.clamp(tonumber(p.s) or 0, 0, 500),
            }
        end
    end
    return out
end

-- distance (flat) from point p to segment a-b
local function distToSegment(p, a, b)
    local abx, abz = b.x - a.x, b.z - a.z
    local len2 = abx * abx + abz * abz
    if len2 == 0 then return math.sqrt((p.x - a.x) ^ 2 + (p.z - a.z) ^ 2) end
    local t = math.clamp(((p.x - a.x) * abx + (p.z - a.z) * abz) / len2, 0, 1)
    return math.sqrt((p.x - (a.x + t * abx)) ^ 2 + (p.z - (a.z + t * abz)) ^ 2)
end

-- Ramer-Douglas-Peucker: turns a wobbly recorded walk into straight lines
local function simplify(list, eps)
    local n = #list
    if n <= 2 then return list end
    local keep = { [1] = true, [n] = true }
    local stack = { { 1, n } }
    while #stack > 0 do
        local seg = table.remove(stack)
        local i1, i2 = seg[1], seg[2]
        local a, b = list[i1], list[i2]
        local maxD, maxI = 0, nil
        for i = i1 + 1, i2 - 1 do
            local d = distToSegment(list[i], a, b)
            if d > maxD then maxD, maxI = d, i end
        end
        if maxI and maxD > eps then
            keep[maxI] = true
            stack[#stack + 1] = { i1, maxI }
            stack[#stack + 1] = { maxI, i2 }
        end
    end
    local out = {}
    for i = 1, n do
        if keep[i] then out[#out + 1] = list[i] end
    end
    return out
end

-- Moving-average smoothing of the recorded drive path (removes the wiggles from your own
-- manual steering, so the target direction doesn't jump around). Endpoints are kept exact.
local function smoothDrive(list)
    local n = #list
    if n < 5 then return list end
    local len = 0
    for i = 2, n do
        len += (Vector3.new(list[i].x - list[i - 1].x, 0, list[i].z - list[i - 1].z)).Magnitude
    end
    local spacing = len / (n - 1)
    local radius = math.floor(DRIVE_SMOOTH_STUDS / math.max(spacing, 0.1) + 0.5)
    if radius < 1 then return list end

    local out = {}
    for i = 1, n do
        if i == 1 or i == n then
            out[i] = list[i]
        else
            local sx, sy, sz, ss, c = 0, 0, 0, 0, 0
            for j = math.max(1, i - radius), math.min(n, i + radius) do
                local p = list[j]
                sx += p.x; sy += p.y; sz += p.z; ss += p.s; c += 1
            end
            out[i] = { x = sx / c, y = sy / c, z = sz / c, s = ss / c }
        end
    end
    return out
end

local function parsePath(text)
    text = text:gsub("^\239\187\191", "") -- strip UTF-8 BOM if present
    local ok, data = pcall(HttpService.JSONDecode, HttpService, text)
    if not ok or type(data) ~= "table" then return nil end
    local drive = smoothDrive(clean(data.drive or data.points)) -- "points" = old recorder format
    if #drive < 2 then return nil end
    local walk = simplify(clean(data.walk), WALK_SIMPLIFY)

    local cum = { 0 } -- cumulative path length, used to avoid exiting early on looping routes
    for i = 2, #drive do
        cum[i] = cum[i - 1] + (flat(pt(drive, i)) - flat(pt(drive, i - 1))).Magnitude
    end
    return {
        drive = drive, walk = walk, cum = cum,
        exits = ALWAYS_EXIT_AT_END or data.exited == true or #walk > 0,
    }
end

-- Repo loading: fetches BASE_URL/1.txt, 2.txt, ... directly
----------------------------------------------------------------
local httpRequest = (syn and syn.request) or (http and http.request)
    or http_request or request or (fluxus and fluxus.request)

local lastHttpStatus = "no http function"
local function httpGet(url)
    if httpRequest then
        local ok, res = pcall(httpRequest, { Url = url, Method = "GET" })
        if ok and res then
            lastHttpStatus = "HTTP " .. tostring(res.StatusCode)
            if res.StatusCode == 200 then return res.Body end
            return nil
        end
    end
    local ok, body = pcall(game.HttpGet, game, url)
    if ok and type(body) == "string" then
        lastHttpStatus = "HTTP 200"
        return body
    end
    return nil
end

local repoPaths = {}   -- [n] = parsed path
local nextProbe = 1    -- next number to look for (so new files are found on later starts)

local function refreshRepo()
    local misses, n = 0, nextProbe
    while misses < MISS_LIMIT and n <= MAX_FILES do
        local body = httpGet(BASE_URL .. "/" .. n .. FILE_EXT)
        local p = body and parsePath(body)
        if p then
            repoPaths[n] = p
            nextProbe = n + 1
            misses = 0
        else
            misses += 1
        end
        n += 1
    end
end

local function loadRandomPath()
    refreshRepo()
    local keys = {}
    for k in pairs(repoPaths) do keys[#keys + 1] = k end
    if #keys > 0 then
        local k = keys[math.random(#keys)]
        return repoPaths[k], k .. FILE_EXT
    end

    if ALLOW_LOCAL_FALLBACK then
        local ok, text = pcall(readfile, LOCAL_FALLBACK)
        local p = ok and parsePath(text)
        if p then return p, LOCAL_FALLBACK end
    end
    return nil, nil, ("Repo failed (%s). Check BASE_URL"):format(lastHttpStatus)
end

-- phases: idle, loading, drive, exiting, walk
local phase = "idle"
local path, pathName
local idx, widx = 1, 1
local stuckT = 0
local stallT = 0
local prevHeading, prevSrc, yawRate = nil, nil, 0
local walkTarget, lastMoveTo = nil, 0

local function stop(msg)
    dbg.Text = ""
    phase = "idle"
    releaseAll()
    local hum, root = getHumanoid(), getRoot()
    if hum and root then pcall(function() hum:MoveTo(root.Position) end) end
    label.Text = msg or "Stopped"
end

local function exitCar()
    phase = "exiting"
    task.spawn(function()
        -- Direction of travel when braking starts. S is only held while the car still moves along
        -- it, so braking can never turn into reversing.
        local part0 = getVehiclePart()
        local v0 = part0 and flat(part0.AssemblyLinearVelocity) or Vector3.zero
        local dir0 = v0.Magnitude > 1 and v0.Unit or nil

        local function brake()
            local t0 = os.clock()
            while phase == "exiting" and os.clock() - t0 < 4 do
                local part = getVehiclePart()
                local v = part and flat(part.AssemblyLinearVelocity) or Vector3.zero
                local along = dir0 and v:Dot(dir0) or 0
                if along < 2 then break end
                setKey(KC.W, false); setKey(KC.A, false); setKey(KC.D, false)
                setKey(KC.S, true)
                label.Text = "Braking..."
                task.wait(0.03)
            end
            releaseAll()
            -- let any remaining roll settle before pressing F
            local t1 = os.clock()
            while phase == "exiting" and os.clock() - t1 < 1.5 do
                local part = getVehiclePart()
                if not part or flat(part.AssemblyLinearVelocity).Magnitude < 1.5 then break end
                task.wait(0.05)
            end
        end

        -- press F to leave the car (re-stop and retry if still seated)
        for _ = 1, 3 do
            if phase ~= "exiting" then return end
            brake()
            if phase ~= "exiting" then return end
            label.Text = "Pressing F to exit..."
            tapKey(EXIT_KEY)
            task.wait(1.5)
            if not isInCar() then break end
        end
        if phase ~= "exiting" then return end

        if #path.walk == 0 then return stop("Finished (no walk path)") end
        widx, stuckT, walkTarget = 1, 0, nil
        phase = "walk"
    end)
end

local userStopped = false

local function start()
    if phase ~= "idle" then return false end
    phase = "loading"
    label.Text = "Loading paths from repo..."
    local p, name, err = loadRandomPath()
    if phase ~= "loading" then return false end -- cancelled while loading
    if not p then stop(err or "No path found") return false end
    local part = getVehiclePart()
    if not part then stop("No vehicle found") return false end

    path, pathName = p, name
    -- start at the waypoint closest to the car
    local pos = part.Position
    local best, bestD = 1, math.huge
    for i = 1, #path.drive do
        local d = (flat(pt(path.drive, i)) - flat(pos)).Magnitude
        if d < bestD then best, bestD = i, d end
    end
    idx = best
    stallT = 0
    prevHeading, prevSrc, yawRate = nil, nil, 0
    phase = "drive"
    return true
end

UserInput.InputBegan:Connect(function(input, _)
    if input.KeyCode ~= TOGGLE_KEY or UserInput:GetFocusedTextBox() then return end
    if phase == "idle" then
        userStopped = false
        start()
    else
        userStopped = true
        stop("Stopped")
    end
end)

-- Pulse a key pair to approximate analog input: v in [-1, 1]
local function drivePWM(v, posKey, negKey)
    local phasePos = (os.clock() % PWM_PERIOD) / PWM_PERIOD
    local on = math.abs(v) > phasePos
    setKey(posKey, v > 0 and on)
    setKey(negKey, v < 0 and on)
end

local fwdLocal = nil -- the car's forward axis in its own space; learned while driving forward

local function unitOr(v, fallback)
    return v.Magnitude > 0.01 and v.Unit or fallback
end

local function driveStep(dt)
    local pts = path.drive
    local part = getVehiclePart()
    if not part then return stop("Lost vehicle") end

    local pos = part.Position
    local vel = flat(part.AssemblyLinearVelocity)
    local speed = vel.Magnitude

    local lookahead = LOOKAHEAD_MIN + speed * LOOKAHEAD_SPEED
    while idx < #pts and (flat(pt(pts, idx)) - flat(pos)).Magnitude < lookahead do
        idx += 1
    end

    -- within EXIT_RADIUS of the last drive waypoint (and near the end of the route): leave the car
    local distLast = (flat(pt(pts, #pts)) - flat(pos)).Magnitude
    if distLast < EXIT_RADIUS and (path.cum[#pts] - path.cum[idx]) < EXIT_RADIUS * 3 then
        releaseAll()
        if path.exits then return exitCar() end
        return stop("Finished")
    end

    local dir = flat(pt(pts, idx) - pos)
    if dir.Magnitude < 0.1 then return end
    dir = dir.Unit

    -- SIGNED speed toward the route target. Negative = rolling backwards / away from it.
    -- (Plain speed can't tell forward from reverse, which made the old code keep holding S
    -- while the car reversed away.)
    local along = vel:Dot(dir)

    -- 1) Car heading from its CFrame. The forward direction (in the part's local space) is learned
    --    while driving straight, so any axis / yaw offset of the part is handled automatically.
    local forward, src
    if fwdLocal then
        forward = flat(part.CFrame:VectorToWorldSpace(fwdLocal))
        forward = forward.Magnitude > 0.01 and forward.Unit or nil
        src = "cf"
        -- sanity check against the real direction of travel
        if forward and along > 6 and forward:Dot(vel.Unit) < 0.5 then forward, src = nil, nil end
    end
    if not forward then
        if along > 3 then
            forward, src = vel.Unit, "vel"
        else
            forward, src = unitOr(flat(part.CFrame.LookVector), dir), "look"
        end
    end

    -- 2) Yaw rate of that heading (positive = turning left)
    local heading = math.atan2(-forward.X, -forward.Z)
    if prevHeading and prevSrc == src and dt > 0 then
        local d = (heading - prevHeading + math.pi) % (2 * math.pi) - math.pi
        yawRate = yawRate * 0.5 + (d / dt) * 0.5
    else
        yawRate = 0
    end
    prevHeading, prevSrc = heading, src

    -- 3) Learn the forward axis while driving straight toward the target
    if along > 8 and along > speed * 0.9 and math.abs(yawRate) < 0.3 then
        local rel = part.CFrame:VectorToObjectSpace(vel.Unit)
        local flatRel = Vector3.new(rel.X, 0, rel.Z)
        if flatRel.Magnitude > 0.7 then
            local u = flatRel.Unit
            fwdLocal = fwdLocal and unitOr(fwdLocal:Lerp(u, 0.1), u) or u
        end
    end

    -- 4) Steering: match the car's heading to the direction of the route ahead.
    --    Decisive proportional control (full lock at ~10 degrees of error) minus a lead term from the
    --    current yaw rate, so it lets go of the key early enough to not overshoot.
    local angle = math.atan2(forward:Cross(dir).Y, forward:Dot(dir)) -- positive = target on the left
    local steer = math.clamp((angle / STEER_FULL_ANGLE - STEER_LEAD * yawRate) * 1.5, -1, 1)
    if math.abs(steer) < 0.1 then steer = 0 end
    drivePWM(steer, KC.A, KC.D) -- left = A, right = D
    dbg.Text = ("heading:%s  err:%+.0fdeg  yaw:%+.2f rad/s  steer:%+.2f"):format(
        src, math.deg(angle), yawRate, steer)

    -- speed control on the signed speed: brake (S) ONLY while really moving toward the target
    -- faster than wanted. If we are rolling backwards, the error is large and W is pressed instead.
    local targetSpeed = pts[idx].s * SPEED_SCALE
    local throttle = math.clamp((targetSpeed - along) / SPEED_RANGE, -1, 1)
    if throttle < 0 and along < 3 then throttle = 0 end

    -- anti-stall: if the car has been (almost) stopped for a while, floor it until it moves
    if speed < STALL_SPEED then stallT += dt else stallT = 0 end
    if stallT > STALL_TIME then throttle = 1 end

    drivePWM(throttle, KC.W, KC.S)

    local keys = (held[KC.W] and "W" or "-") .. (held[KC.A] and "A" or "-")
        .. (held[KC.S] and "S" or "-") .. (held[KC.D] and "D" or "-")
    label.Text = ("DRIVE %d/%d | %.0f(%.0f)/%.0f studs/s | %.0f to end | %s | %s | %s"):format(
        idx, #pts, along, speed, targetSpeed, distLast, keys,
        isInCar() and "inCar" or "carUnknown", tostring(pathName))
end

local function walkStep(dt)
    local hum, root = getHumanoid(), getRoot()
    if not hum or not root then return end

    local walkPts = path.walk
    local pos = root.Position

    -- skip waypoints already reached (the final one uses a tighter radius)
    while widx <= #walkPts do
        local reach = (widx == #walkPts) and 2.5 or WALK_REACH
        if (flat(pt(walkPts, widx)) - flat(pos)).Magnitude < reach then
            widx += 1
            walkTarget = nil
        else
            break
        end
    end
    if widx > #walkPts then return stop("Arrived at destination") end

    -- only re-issue MoveTo when the target changes (or every second), so movement stays smooth
    local target = pt(walkPts, widx)
    if target ~= walkTarget or os.clock() - lastMoveTo > 1 then
        hum:MoveTo(target)
        if target ~= walkTarget then stuckT = 0 end
        walkTarget, lastMoveTo = target, os.clock()
    end

    -- jump if blocked
    if flat(root.AssemblyLinearVelocity).Magnitude < 1 then
        stuckT += dt
        if stuckT > 1.5 then hum.Jump = true stuckT = 0 end
    else
        stuckT = 0
    end
    label.Text = ("WALK %d/%d | %s"):format(widx, #walkPts, tostring(pathName))
end

local function wantSprint()
    if not ALWAYS_SPRINT_OUT_OF_CAR then return false end
    if phase == "drive" or phase == "exiting" then return false end
    return getHumanoid() ~= nil and not isInCar()
end

RunService.Heartbeat:Connect(function(dt)
    setKey(KC.LeftShift, wantSprint()) -- hold SHIFT whenever you're out of the car

    if phase == "drive" then
        driveStep(dt)
        refreshKeys()
    elseif phase == "exiting" then
        refreshKeys()
    elseif phase == "walk" then
        walkStep(dt)
        refreshKeys()
    end
end)

-- Auto-start: begin as soon as the script runs (waits until you are in a car if needed).
if AUTO_START then
    task.spawn(function()
        if not game:IsLoaded() then game.Loaded:Wait() end
        while not getRoot() do task.wait(0.5) end
        while not userStopped and phase == "idle" do
            if not AUTO_START_REQUIRE_IN_CAR or isInCar() then
                if start() then return end
            else
                label.Text = "Waiting until you are in a car..."
            end
            task.wait(2)
        end
    end)
end
